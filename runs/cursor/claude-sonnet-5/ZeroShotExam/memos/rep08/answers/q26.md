func (s *Store) DeleteMemo(ctx context.Context, delete *DeleteMemo) error {
	// Clean up memo_relation records where this memo is either the source or target.
	if err := s.driver.DeleteMemoRelation(ctx, &DeleteMemoRelation{MemoID: &delete.ID}); err != nil {
		return err
	}
	if err := s.driver.DeleteMemoRelation(ctx, &DeleteMemoRelation{RelatedMemoID: &delete.ID}); err != nil {
		return err
	}
	// Clean up attachments linked to this memo.
	attachments, err := s.ListAttachments(ctx, &FindAttachment{MemoID: &delete.ID})
	if err != nil {
		return err
	}
	for _, attachment := range attachments {
		if err := s.DeleteAttachment(ctx, &DeleteAttachment{ID: attachment.ID}); err != nil {
			return err
		}
	}
	return s.driver.DeleteMemo(ctx, delete)
}

- **`memo_share`** — not touched by any Go code, but each backend schema declares `memo_share.memo_id` with `FOREIGN KEY (memo_id) REFERENCES memo(id) ON DELETE CASCADE`, so removing the `memo` row cascades to delete its shares at the database level (SQLite: `/bench/repo/store/migration/sqlite/LATEST.sql:109`; Postgres: `/bench/repo/store/migration/postgres/LATEST.sql:108`; MySQL: `/bench/repo/store/migration/mysql/LATEST.sql:108`). Note: SQLite's connection is opened with `_pragma=foreign_keys(0)` (`/bench/repo/store/db/sqlite/sqlite.go:53`), so this cascade is effectively disabled for the SQLite backend, but is enforced for Postgres/MySQL.

Not affected: **`reaction`** is read (`ListReactions`) only to build the webhook payload before deletion, but `DeleteMemo` never calls `Store.DeleteReaction`, and `reaction.content_id` is a plain text column with no foreign key to `memo`, so reaction rows are left orphaned rather than deleted (`/bench/repo/server/router/api/v1/memo_service.go:545-550`, `/bench/repo/store/migration/sqlite/LATEST.sql:92-99`).
